# 2. FULL-STACK APPLICATION & CLINICAL INTERFACE
# ==============================================================================

# ------------------------------------------------------------------------------
# Option A — Decoupled Development Mode (Hot-Reloading):
# ------------------------------------------------------------------------------
# Terminal 1 (FastAPI Deep Learning Backend):
.\venv\Scripts\python.exe -m uvicorn backend.app:app --host 127.0.0.1 --port 8000 --reload

# Terminal 2 (React / Vite Frontend):
npm.cmd --prefix frontend run dev

# -> Access Frontend UI: http://localhost:3000
# -> Access Backend Swagger API Docs: http://localhost:8000/docs


# ------------------------------------------------------------------------------
# Option B — Single-Command Unified Serving (FastAPI serving pre-built UI):
# ------------------------------------------------------------------------------
# Build frontend assets:
cd frontend; npm.cmd run build; cd ..

# Launch production server:
.\venv\Scripts\python.exe -m uvicorn backend.app:app --host 127.0.0.1 --port 8000

# -> Access Unified Dashboard: http://localhost:8000


# ------------------------------------------------------------------------------
# Option C — Containerized Multi-Service Deployment (Docker Compose):
# ------------------------------------------------------------------------------
# 1. GPU Acceleration Mode (Requires NVIDIA GPU & Container Toolkit):
docker compose up --build -d

# 2. CPU Mode (Standard multi-core CPU with bfloat16 acceleration):
docker compose -f docker-compose.cpu.yml up --build -d

# -> Access Frontend UI: http://localhost:3000
# -> Access Backend API Docs: http://localhost:8000/docs

# Check container status & health:
docker compose ps
docker compose logs -f backend

# Benchmark inference speed inside container:
docker compose exec backend python -m backend.benchmark_inference --case-id case_00003 --compare-fp32

# Stop and clean up containers:
# For GPU mode:
docker compose down

# For CPU mode:
docker compose -f docker-compose.cpu.yml down


# ==============================================================================
# 3. UNIT TESTING & HEALTH VERIFICATION
# ==============================================================================
# Run backend test suite:
.\venv\Scripts\python.exe -m pytest backend/tests

# Verify backend health and model loading:
curl http://localhost:8000/api/health
